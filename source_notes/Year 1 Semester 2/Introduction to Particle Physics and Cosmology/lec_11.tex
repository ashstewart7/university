\[
    a(t) = \left(\frac{t}{t_0}\right)^{2/3}

\[
    \rho_m(t) = \frac{\rho_{m, 0}}{a(t)^3} = \rho_{m, 0} \left(\frac{t_0}{t}\right)^2
\]
